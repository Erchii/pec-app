\section{Running the Application}

\begin{lstlisting}[language=bash,caption={Setup and launch instructions}]
# 1. Install dependencies
pip install flask flask-cors pandas numpy scikit-learn xgboost joblib

# 2. Augment dataset
python expand_dataset.py

# 3. Train models on augmented dataset
cd backend
python train.py --data data/pec_dataset.csv --cv 5

# 4. Start API server (http://localhost:5001)
python app.py

# The deployed application is accessible at:
# http://178.105.34.179/
\end{lstlisting}
